async function locate() { const ip = document.getElementById('ipInput').value.trim(); if (!ip) return alert('Masukkan IP!'); document.getElementById('result').textContent = '⏳ Melacak...'; try { // Pakai ipapi.co — biasanya tidak diblokir const res = await fetch(`https://ipapi.co/${ip}/json/`); const data = await res.json(); if (data.error) { document.getElementById('result').textContent = `❌ ${data.error}`; return; } displayData({ ip: data.ip, country: data.country_name, region: data.region, city: data.city, latitude: data.latitude, longitude: data.longitude, org: data.org, isp: data.org, asn: data.asn, timezone: data.timezone }); } catch (error) { document.getElementById('result').textContent = '❌ Gagal melacak. Coba lagi.'; } }
